```typescript
export const LocationExtractionSchema = z.object({
  addresses: z.array(z.object({
    text: z.string().describe("Full address as found in document"),
    context: z.enum(["title_block", "body", "footer", "unknown"])
      .describe("Where in the document this address was found"),
    isProjectSite: z.boolean()
      .describe("True if this appears to be the construction site address"),
  })),
  intersections: z.array(z.object({
    road1: z.string(),
    road2: z.string(),
  })),
  projectName: z.string().nullable(),
  city: z.string().nullable(),
  state: z.string().nullable(),
});

export type LocationExtraction = z.infer<typeof LocationExtractionSchema>;
```

3. **extractLocationsFromPDF function**:
```typescript
export async function extractLocationsFromPDF(
  ai: GoogleGenAI,
  pdfPath: string
): Promise<LocationExtraction> {
  // Upload PDF to Files API for large files
  const uploadedFile = await ai.files.upload({
    file: pdfPath,
    config: { mimeType: "application/pdf" },
  });

  const response = await ai.models.generateContent({
    model: "gemini-2.5-flash",
    contents: createUserContent([
      createPartFromUri(uploadedFile.uri!, uploadedFile.mimeType!),
      `Extract ALL addresses and location signals from this construction/grading plan PDF.

Look for:
- Addresses in the title block (usually the project site)
- Addresses elsewhere (may be contractor, engineer, or material suppliers)
- Street intersections mentioned (e.g., "NE corner of Bell Rd and 99th Ave")
- Project name
- City and state

The title block address is most likely the actual construction site.
Mark isProjectSite=true for addresses that appear to be the site location.`,
    ]),
    config: {
      responseMimeType: "application/json",
      responseSchema: z.toJSONSchema(LocationExtractionSchema),
    },
  });

  // Cleanup uploaded file
  await ai.files.delete({ name: uploadedFile.name! });

  return LocationExtractionSchema.parse(JSON.parse(response.text ?? "{}"));
}
```

**CRITICAL:** Use `z.toJSONSchema()` (Zod v4 native), NOT `zodToJsonSchema()` from zod-to-json-schema. The latter is broken with Zod v4 and returns incomplete schemas.

**Note:** The config property is `responseSchema` not `responseJsonSchema` per current @google/genai API.

2. **Google Geocoding API function** (primary):
```typescript
const GOOGLE_GEOCODE_URL = "https://maps.googleapis.com/maps/api/geocode/json";

export async function geocodeAddress(
  address: string,
  apiKey: string
): Promise<GeoCoord | null> {
  try {
    const params = new URLSearchParams({
      address,
      key: apiKey,
    });

    const response = await fetch(`${GOOGLE_GEOCODE_URL}?${params}`);
    const data = await response.json();

    if (data.status !== "OK" || !data.results?.[0]?.geometry?.location) {
      return null;
    }

    const { lat, lng } = data.results[0].geometry.location;
    return geoCoord(lat, lng);
  } catch {
    return null;
  }
}
```

3. **Nominatim fallback** (free, rate-limited):
```typescript
const NOMINATIM_URL = "https://nominatim.openstreetmap.org/search";

export async function geocodeAddressNominatim(
  address: string
): Promise<GeoCoord | null> {
  try {
    const params = new URLSearchParams({
      q: address,
      format: "json",
      limit: "1",
    });

    const response = await fetch(`${NOMINATIM_URL}?${params}`, {
      headers: { "User-Agent": "auto-permit/1.0" },
    });
    const data = await response.json();

    if (!data?.[0]?.lat || !data?.[0]?.lon) {
      return null;
    }

    return geoCoord(Number(data[0].lat), Number(data[0].lon));
  } catch {
    return null;
  }
}
```

4. **Geocode with fallback chain**:
```typescript
export async function geocodeWithFallback(
  address: string,
  googleApiKey?: string
): Promise<GeoCoord | null> {
  // Try Google first if API key available
  if (googleApiKey) {
    const result = await geocodeAddress(address, googleApiKey);
    if (result) return result;
  }

  // Fallback to Nominatim
  return geocodeAddressNominatim(address);
}
```

5. **Intersection geocoding**:
```typescript
export async function geocodeIntersection(
  road1: string,
  road2: string,
  city?: string,
  googleApiKey?: string
): Promise<GeoCoord | null> {
  const query = city
    ? `${road1} and ${road2}, ${city}`
    : `${road1} and ${road2}`;

  return geocodeWithFallback(query, googleApiKey);
}
```

6. **Create signal from address**:
```typescript
export async function createSignalFromAddress(
  address: string,
  source: SignalSource,
  googleApiKey?: string
): Promise<LocationSignal> {
  const coords = await geocodeWithFallback(address, googleApiKey);

  return {
    source,
    query: address,
    coords,
    confidence: coords ? SIGNAL_WEIGHTS[source] : 0,
  };
}
```

7. **Create signal from intersection**:
```typescript
export async function createSignalFromIntersection(
  road1: string,
  road2: string,
  city?: string,
  googleApiKey?: string
): Promise<LocationSignal> {
  const coords = await geocodeIntersection(road1, road2, city, googleApiKey);
  const query = city ? `${road1} & ${road2}, ${city}` : `${road1} & ${road2}`;

  return {
    source: "intersection",
    query,
    coords,
    confidence: coords ? SIGNAL_WEIGHTS.intersection : 0,
  };
}
```

**Key design decisions:**
- Never throw on geocoding failure - return null coords with 0 confidence
- Google API key is optional - falls back to Nominatim if not provided
- Each signal includes the original query for debugging
