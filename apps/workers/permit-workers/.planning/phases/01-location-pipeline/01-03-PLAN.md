```typescript
function calculateCentroid(coords: GeoCoord[]): GeoCoord {
  const sumLat = coords.reduce((sum, c) => sum + c.lat, 0);
  const sumLng = coords.reduce((sum, c) => sum + c.lng, 0);
  return geoCoord(sumLat / coords.length, sumLng / coords.length);
}
```

4. **Max radius calculation helper**:
```typescript
function calculateMaxRadius(centroid: GeoCoord, signals: LocationSignal[]): number {
  if (signals.length === 0) return 0;
  return Math.max(
    ...signals.map((s) => haversineDistance(centroid, s.coords!))
  );
}
```

5. **Main clustering function** (greedy algorithm from research):
```typescript
export function clusterSignals(
  signals: LocationSignal[]
): { clusters: LocationCluster[]; outliers: LocationSignal[] } {
  // Filter to only signals with valid coordinates
  const validSignals = signals.filter((s) => s.coords !== null);
  const clusters: LocationCluster[] = [];
  const used = new Set<number>();

  for (let i = 0; i < validSignals.length; i++) {
    if (used.has(i)) continue;

    const signal = validSignals[i]!;
    const clusterSignals: LocationSignal[] = [signal];
    used.add(i);

    // Find all signals within radius of this signal
    for (let j = i + 1; j < validSignals.length; j++) {
      if (used.has(j)) continue;

      const otherSignal = validSignals[j]!;
      const distance = haversineDistance(signal.coords!, otherSignal.coords!);

      if (distance <= CLUSTER_RADIUS_METERS) {
        clusterSignals.push(otherSignal);
        used.add(j);
      }
    }

    // Calculate cluster properties
    const centroid = calculateCentroid(clusterSignals.map((s) => s.coords!));
    const totalConfidence = clusterSignals.reduce((sum, s) => sum + s.confidence, 0);

    clusters.push({
      centroid,
      signals: clusterSignals,
      radiusMeters: calculateMaxRadius(centroid, clusterSignals),
      totalConfidence,
    });
  }

  // Sort clusters by total confidence (best first)
  clusters.sort((a, b) => b.totalConfidence - a.totalConfidence);

  // Signals without coords are outliers
  const outliers = signals.filter((s) => s.coords === null);

  return { clusters, outliers };
}
```

**Key design:**
- Uses simple greedy clustering (not DBSCAN) - appropriate for 3-10 signals
- 500m radius is wide enough to catch same-block variations but narrow enough to exclude contractor addresses
- Clusters sorted by total confidence so primary is always first

```typescript
export async function runLocationPipeline(
  options: PipelineOptions
): Promise<LocationPipelineResult> {
  const { pdfPath, geminiApiKey, googleMapsApiKey, includeParcelLookup = true, verbose = false } = options;
  const processingLog: string[] = [];

  const addLog = (msg: string) => {
    processingLog.push(msg);
    log(verbose, msg);
  };

  addLog("Starting location pipeline");

  // Step 1: Extract locations from PDF
  addLog("Extracting locations from PDF...");
  const ai = new GoogleGenAI({ apiKey: geminiApiKey });
  const extraction = await extractLocationsFromPDF(ai, pdfPath);
  addLog(`Found ${extraction.addresses.length} addresses, ${extraction.intersections.length} intersections`);

  // Step 2: Create signals from extracted data
  addLog("Geocoding extracted locations...");
  const signals: LocationSignal[] = [];

  // Geocode addresses
  for (const addr of extraction.addresses) {
    const source = addr.context === "title_block" ? "address_title_block" : "address_body";
    const signal = await createSignalFromAddress(addr.text, source, googleMapsApiKey);
    signals.push(signal);
    addLog(`  ${addr.text} -> ${signal.coords ? 'OK' : 'FAILED'}`);
  }

  // Geocode intersections
  for (const int of extraction.intersections) {
    const signal = await createSignalFromIntersection(
      int.road1,
      int.road2,
      extraction.city ?? undefined,
      googleMapsApiKey
    );
    signals.push(signal);
    addLog(`  ${int.road1} & ${int.road2} -> ${signal.coords ? 'OK' : 'FAILED'}`);
  }

  addLog(`Total signals: ${signals.length}, with coords: ${signals.filter(s => s.coords).length}`);

  // Step 3: Cluster signals
  addLog("Clustering signals...");
  const { clusters, outliers } = clusterSignals(signals);
  addLog(`Formed ${clusters.length} clusters, ${outliers.length} outliers`);

  // Step 4: Determine primary cluster and consensus
  const primaryCluster = clusters[0] ?? null;
  const consensusLocation = primaryCluster?.centroid ?? null;

  // Calculate overall confidence (normalized to 0-1)
  // Formula: primary cluster confidence / max possible confidence
  const maxPossibleConfidence = signals.length > 0
    ? signals.reduce((sum, s) => sum + (s.coords ? s.confidence : 0), 0)
    : 1;
  const consensusConfidence = primaryCluster
    ? Math.min(primaryCluster.totalConfidence / Math.max(maxPossibleConfidence, 1), 1)
    : 0;

  addLog(`Consensus location: ${consensusLocation?.lat.toFixed(4)}, ${consensusLocation?.lng.toFixed(4)}`);
  addLog(`Consensus confidence: ${(consensusConfidence * 100).toFixed(1)}%`);

  // Step 5: Parcel lookup (optional)
  let parcel: ParcelData | null = null;
  if (includeParcelLookup && consensusLocation) {
    addLog("Looking up parcel...");
    try {
      parcel = await queryParcelByCoordinates(consensusLocation.lat, consensusLocation.lng);
      if (parcel) {
        addLog(`Found parcel: ${parcel.apn} - ${parcel.address}`);
      }
    } catch (err) {
      addLog(`Parcel lookup failed: ${err}`);
    }
  }

  return {
    signals,
    clusters,
    primaryCluster,
    outliers,
    consensusLocation,
    consensusConfidence,
    parcel,
    processingLog,
  };
}
```
